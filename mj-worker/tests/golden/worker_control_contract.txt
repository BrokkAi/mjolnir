== script: liveness
hel_root='<ROOT>'
if [ -e "$hel_root/hel" ]; then
    hel_bin=
    for hel_candidate in "$hel_root/hel" "$hel_root"/hel-control*; do
        if [ -f "$hel_candidate" ] && [ -x "$hel_candidate" ] \
            && "$hel_candidate" worker liveness --help >/dev/null 2>&1; then
            hel_bin=$hel_candidate
            break
        fi
    done
    if [ -z "$hel_bin" ]; then
        echo "no worker in $hel_root serves 'worker liveness'" >&2
        exit 78
    fi
    "$hel_bin" worker liveness --root "$hel_root" || exit $?
else
    printf 'dead\n'
fi
